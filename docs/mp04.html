slow_download <- purrr::slowly(download.file, purrr::rate_delay(pause = 5))
olympedia_download_counter <- 0L

read_olympedia <- function(url) {
  if (stringr::str_starts(url, "/")) {
    url <- stringr::str_remove(url, "^/")
  }

  cache_path <- fs::path(
    cache_dir,
    url |>
      as.character() |>
      stringr::str_replace_all("[/?=&]+", "_")
  )

  if (!fs::file_exists(cache_path)) {
    src_url <- paste0("https://www.olympedia.org/", url)
    olympedia_download_counter <<- olympedia_download_counter + 1L
    message(
      "Downloading uncached Olympedia page ", olympedia_download_counter,
      ": ", src_url
    )
    slow_download(src_url, cache_path, mode = "wb", quiet = TRUE)
  }

  rvest::read_html(cache_path)
}

extract_country_code_from_img <- function(node) {
  src <- node |>
    rvest::html_elements("img") |>
    rvest::html_attr("src")

  code <- src |>
    stringr::str_extract("[A-Z]{3}(?=\\.png)") |>
    stats::na.omit()

  if (length(code) == 0) NA_character_ else code[[1]]
}

parse_games_for_season <- function(doc, season_name) {
  table_node <- doc |>
    rvest::html_element(
      xpath = glue("//h3[normalize-space() = '{season_name}']/following-sibling::table[1]")
    )

  if (length(table_node) == 0 || is.na(table_node)) {
    return(tibble())
  }

  rows <- table_node |>
    rvest::html_elements("tr")

  purrr::map_dfr(rows[-1], function(row) {
    cells <- row |>
      rvest::html_elements("td")

    cell_text <- cells |>
      rvest::html_text2()

    if (length(cell_text) < 3) {
      return(tibble())
    }

    # Mainline rows generally have: #, Year, City, Country, Opened, Closed, Competition.
    # Some future/not-held rows are incomplete; those get filtered below.
    edition_url <- row |>
      rvest::html_element("a[href^='/editions/']") |>
      rvest::html_attr("href")

    edition_id <- edition_url |>
      stringr::str_extract("\\d+$") |>
      readr::parse_integer()

    year <- cell_text[[2]] |>
      readr::parse_number()

    row_text <- paste(cell_text, collapse = " ")

    tibble(
      edition_id = edition_id,
      edition_roman = cell_text[[1]],
      edition_url = edition_url,
      year = year,
      city = cell_text[[3]],
      host_country_iso = extract_country_code_from_img(row),
      season = season_name,
      row_text = row_text
    )
  }) |>
    filter(
      !is.na(edition_id),
      !is.na(year),
      !str_detect(row_text, regex("not held|cancelled", ignore_case = TRUE)),
      year <= 2026
    ) |>
    mutate(
      edition_integer = suppressWarnings(as.integer(as.roman(edition_roman)))
    ) |>
    select(-row_text)
}

parse_games <- function() {
  editions_doc <- read_olympedia("editions")

  bind_rows(
    parse_games_for_season(editions_doc, "Summer"),
    parse_games_for_season(editions_doc, "Winter")
  ) |>
    distinct(edition_id, .keep_all = TRUE) |>
    arrange(year, season)
}

parse_disciplines_for_edition <- function(edition_id, edition_url) {
  doc <- read_olympedia(edition_url)

  # Keep links to sport pages for this edition. If "Other Disciplines" links are
  # included, the medal-table parser returns zero rows for non-medal pages.
  discipline_links <- doc |>
    rvest::html_elements(glue("a[href^='/editions/{edition_id}/sports/']"))

  tibble(
    edition_id = edition_id,
    discipline_name = discipline_links |> rvest::html_text2(),
    discipline_url = discipline_links |> rvest::html_attr("href")
  ) |>
    filter(!is.na(discipline_url), discipline_name != "") |>
    mutate(
      discipline_code = stringr::str_extract(discipline_url, "[A-Z0-9]+$")
    ) |>
    distinct(edition_id, discipline_code, .keep_all = TRUE)
}

parse_disciplines <- function(games) {
  games |>
    select(edition_id, edition_url) |>
    mutate(discipline_data = map2(edition_id, edition_url, parse_disciplines_for_edition)) |>
    select(discipline_data) |>
    unnest(discipline_data)
}

parse_medal_table_for_discipline <- function(discipline_url) {
  doc <- read_olympedia(discipline_url)

  medal_table_node <- doc |>
    rvest::html_element(
      xpath = "//h2[normalize-space() = 'Medal table']/following-sibling::table[1]"
    )

  if (length(medal_table_node) == 0 || is.na(medal_table_node)) {
    return(tibble())
  }

  rows <- medal_table_node |>
    rvest::html_elements("tr")

  purrr::map_dfr(rows[-1], function(row) {
    cells <- row |>
      rvest::html_elements("td")

    if (length(cells) < 5) {
      return(tibble())
    }

    noc_cell <- cells[[1]]
    noc_text <- noc_cell |> rvest::html_text2()

    competitor_iso <- noc_cell |>
      rvest::html_elements("a[href^='/countries/']") |>
      rvest::html_attr("href") |>
      stringr::str_extract("[A-Z0-9]{3}$") |>
      stats::na.omit()

    competitor_iso <- if (length(competitor_iso) == 0) {
      stringr::str_extract(noc_text, "[A-Z0-9]{3}$")
    } else {
      competitor_iso[[length(competitor_iso)]]
    }

    medal_counts <- cells[2:5] |>
      rvest::html_text2() |>
      readr::parse_integer(na = c("", "—", "-"))

    tibble(
      competitor_country = noc_text |>
        stringr::str_remove("\\s+[A-Z0-9]{3}$") |>
        stringr::str_squish(),
      competitor_iso = competitor_iso,
      Gold = medal_counts[[1]],
      Silver = medal_counts[[2]],
      Bronze = medal_counts[[3]],
      Total = medal_counts[[4]]
    )
  }) |>
    mutate(across(all_of(medal_types), ~ replace_na(.x, 0L)))
}

parse_medal_table_cached <- function(edition_id, discipline_code, discipline_url) {
  result_cache_path <- fs::path(
    medal_table_cache_dir,
    glue("edition_{edition_id}_sport_{discipline_code}.rds")
  )

  if (!refresh_olympedia_data && fs::file_exists(result_cache_path)) {
    return(readRDS(result_cache_path))
  }

  message("Parsing medal table: edition ", edition_id, " / ", discipline_code)
  result <- parse_medal_table_for_discipline(discipline_url)
  saveRDS(result, result_cache_path)
  result
}

# This is the slowest step on the first run. It is much faster after caching.
# The small per-discipline RDS cache means an interrupted render can resume
# without reparsing medal tables that already succeeded.
if (
  !refresh_olympedia_data &&
    fs::file_exists(games_cache_path) &&
    fs::file_exists(disciplines_cache_path) &&
    fs::file_exists(olympic_results_cache_path)
) {
  games <- readRDS(games_cache_path)
  disciplines <- readRDS(disciplines_cache_path)
  olympic_results <- readRDS(olympic_results_cache_path)
} else {
  if (!refresh_olympedia_data && fs::file_exists(games_cache_path)) {
    games <- readRDS(games_cache_path)
  } else {
    message("Step 1/3: parsing Olympic editions")
    games <- parse_games()
    saveRDS(games, games_cache_path)
  }

  if (!refresh_olympedia_data && fs::file_exists(disciplines_cache_path)) {
    disciplines <- readRDS(disciplines_cache_path)
  } else {
    message("Step 2/3: parsing sport/discipline links for each edition")
    disciplines <- parse_disciplines(games)
    saveRDS(disciplines, disciplines_cache_path)
  }

  message("Step 3/3: parsing medal tables by edition and sport/discipline")
  olympic_results <- disciplines |>
    mutate(
      medal_table = pmap(
        list(edition_id, discipline_code, discipline_url),
        parse_medal_table_cached
      )
    ) |>
    unnest(medal_table) |>
    left_join(games, by = "edition_id") |>
    relocate(
      edition_id, edition_roman, edition_integer, edition_url, year, city,
      host_country_iso, season, discipline_name, discipline_code, discipline_url,
      competitor_country, competitor_iso, Gold, Silver, Bronze, Total
    ) |>
    arrange(year, season, discipline_name, desc(Total), competitor_country)

  saveRDS(olympic_results, olympic_results_cache_path)
  readr::write_csv(olympic_results, fs::path(data_dir, "olympic_results_by_discipline.csv"))
}

standardize_olympedia_iso3 <- function(x) {
  # Converts common Olympedia/historical/non-ISO codes to codes found in
  # modern country-shape data. This is a pragmatic analysis choice; there are
  # defensible alternatives for historical states.
  dplyr::case_match(
    x,
    "GRE" ~ "GRC", # Greece
    "GER" ~ "DEU", # Germany
    "DEN" ~ "DNK", # Denmark
    "URS" ~ "RUS", # Soviet Union -> Russia, per prompt suggestion
    "EUN" ~ "RUS", # Unified Team -> Russia, simplifying assumption
    "ROC" ~ "RUS", # Russian Olympic Committee -> Russia
    "OAR" ~ "RUS", # Olympic Athletes from Russia -> Russia
    "FRG" ~ "DEU", # West Germany -> Germany
    "GDR" ~ "DEU", # East Germany -> Germany
    "TCH" ~ "CZE", # Czechoslovakia -> Czech Republic
    "BOH" ~ "CZE", # Bohemia -> Czech Republic
    "YUG" ~ "SRB", # Yugoslavia -> Serbia, simplifying assumption
    "SCG" ~ "SRB", # Serbia and Montenegro -> Serbia
    "ANZ" ~ "AUS", # Australasia -> Australia, simplifying assumption
    "MIX" ~ NA_character_, # Mixed team: not a country shape
    .default = x
  )
}

pick_existing_col <- function(data, candidates, label) {
  hit <- intersect(candidates, names(data))
  if (length(hit) == 0) {
    stop(
      "Could not find a usable ", label, " column. Available columns are: ",
      paste(names(data), collapse = ", ")
    )
  }
  hit[[1]]
}

edition_country_totals <- olympic_results_clean |>
  group_by(
    edition_id, edition_roman, edition_integer, year, city, season,
    host_country_iso, host_iso3, competitor_iso, competitor_iso3, competitor_country

  summarise(across(all_of(medal_types), ~ sum(.x, na.rm = TRUE)), .groups = "drop")

games_ranked <- games |>
  mutate(host_iso3 = standardize_olympedia_iso3(host_country_iso)) |>
  arrange(season, year) |>
  group_by(season) |>
  mutate(
    prior_edition_id = lag(edition_id),
    prior_year = lag(year)
  ) |>
  ungroup()

country_totals_for_join <- edition_country_totals |>
  select(
    edition_id, season, year, competitor_iso3,
    all_of(medal_types)
  )

host_current_prior <- games_ranked |>
  filter(!is.na(prior_edition_id), year >= recent_start_year) |>
  select(edition_id, prior_edition_id, year, prior_year, season, city, host_iso3) |>
  left_join(
    country_totals_for_join,
    by = c("edition_id", "season", "year", "host_iso3" = "competitor_iso3")
  ) |>
  rename_with(~ paste0("current_", .x), all_of(medal_types)) |>
  left_join(
    country_totals_for_join |>
      rename(prior_edition_id = edition_id, prior_year = year),
    by = c("prior_edition_id", "season", "prior_year", "host_iso3" = "competitor_iso3")
  ) |>
  rename_with(~ paste0("prior_", .x), all_of(medal_types)) |>
  mutate(
    across(starts_with("current_"), ~ replace_na(.x, 0L)),
    across(starts_with("prior_"), ~ replace_na(.x, 0L))
  )

for (m in medal_types) {
  host_current_prior <- host_current_prior |>
    mutate(
      "factor_{m}" := if_else(
        .data[[paste0("prior_", m)]] > 0,
        .data[[paste0("current_", m)]] / .data[[paste0("prior_", m)]],
        NA_real_
      ),
      "pct_change_{m}" := if_else(
        .data[[paste0("prior_", m)]] > 0,
        (.data[[paste0("current_", m)]] - .data[[paste0("prior_", m)]]) /
          .data[[paste0("prior_", m)]],
        NA_real_
      )
    )
}

one_sample_t_ci <- function(data, response_col, conf_level = 0.95) {
  vals <- data[[response_col]]
  vals <- vals[!is.na(vals)]
  n_vals <- length(vals)

  estimate <- if (n_vals > 0) mean(vals) else NA_real_
  sd_val <- if (n_vals > 1) stats::sd(vals) else NA_real_
  se_val <- if (n_vals > 1) sd_val / sqrt(n_vals) else NA_real_

  lower_ci <- NA_real_
  upper_ci <- NA_real_
  statistic <- NA_real_
  t_df <- if (n_vals > 1) n_vals - 1 else NA_real_
  p_value <- NA_real_

  if (n_vals >= 2 && is.finite(sd_val) && sd_val > 0) {
    tt <- stats::t.test(vals, conf.level = conf_level)
    lower_ci <- unname(tt$conf.int[1])
    upper_ci <- unname(tt$conf.int[2])
    statistic <- unname(tt$statistic)
    t_df <- unname(tt$parameter)
    p_value <- tt$p.value
  } else if (n_vals >= 2 && is.finite(sd_val) && sd_val == 0) {
    lower_ci <- estimate
    upper_ci <- estimate
  }

  tibble::tibble(
    estimate = estimate,
    lower_ci = lower_ci,
    upper_ci = upper_ci,
    statistic = statistic,
    t_df = t_df,
    p_value = p_value,
    alternative = "two.sided",
    n = n_vals,
    sd = sd_val,
    se = se_val
  )
}

one_sample_prop_ci_from_counts <- function(successes, trials = NULL, n = NULL, conf_level = 0.95, response_name = NULL, ...) {
  if (is.null(trials)) trials <- n

  successes <- as.numeric(successes)[1]
  trials <- as.numeric(trials)[1]

  if (is.na(successes) || is.na(trials) || trials <= 0) {
    return(tibble::tibble(
      estimate = NA_real_,
      lower_ci = NA_real_,
      upper_ci = NA_real_,
      statistic = NA_real_,
      p_value = NA_real_,
      alternative = "two.sided",
      successes = successes,
      n = trials
    ))
  }

  successes <- max(0, min(successes, trials))
  estimate <- successes / trials

  pt <- tryCatch(
    stats::prop.test(
      x = successes,
      n = trials,
      conf.level = conf_level,
      correct = FALSE
    ),
    error = function(e) NULL
  )

  if (is.null(pt)) {
    return(tibble::tibble(
      estimate = estimate,
      lower_ci = NA_real_,
      upper_ci = NA_real_,
      statistic = NA_real_,
      p_value = NA_real_,
      alternative = "two.sided",
      successes = successes,
      n = trials
    ))
  }

  tibble::tibble(
    estimate = estimate,
    lower_ci = unname(pt$conf.int[1]),
    upper_ci = unname(pt$conf.int[2]),
    statistic = unname(pt$statistic),
    p_value = pt$p.value,
    alternative = pt$alternative,
    successes = successes,
    n = trials
  )
}

sample_from_ci <- function(estimate, lower_ci, upper_ci, n, lower_bound = -Inf, upper_bound = Inf) {
  # Approximate a normal distribution from a 95% CI.
  ci_sd <- (upper_ci - lower_ci) / (2 * qnorm(0.975))

  if (is.na(estimate) || is.na(ci_sd) || ci_sd <= 0) {
    return(rep(NA_real_, n))
  }

  draws <- rnorm(n, mean = estimate, sd = ci_sd)
  pmin(pmax(draws, lower_bound), upper_bound)
}

forecast_2028_medals <- function(n_sim = 100000, new_events = 22) {
  ci_inputs <- us_baseline_ci |>
    select(medal, baseline_estimate = estimate, baseline_lower = lower_ci, baseline_upper = upper_ci) |>
    left_join(
      host_factor_ci |>
        select(medal, host_estimate = estimate, host_lower = lower_ci, host_upper = upper_ci),
      by = "medal"
    ) |>
    left_join(
      new_sport_prob_ci |>
        select(medal, new_estimate = estimate, new_lower = lower_ci, new_upper = upper_ci),
      by = "medal"
    )

  purrr::pmap_dfr(ci_inputs, function(
    medal, baseline_estimate, baseline_lower, baseline_upper,
    host_estimate, host_lower, host_upper,
    new_estimate, new_lower, new_upper
  ) {
    baseline_draw <- sample_from_ci(
      baseline_estimate, baseline_lower, baseline_upper,
      n = n_sim, lower_bound = 0
    )

    host_draw <- sample_from_ci(
      host_estimate, host_lower, host_upper,
      n = n_sim, lower_bound = 0
    )

    new_sport_draw <- sample_from_ci(
      new_estimate, new_lower, new_upper,
      n = n_sim, lower_bound = 0, upper_bound = 1
    )

    forecast_draw <- baseline_draw * host_draw + new_events * new_sport_draw

    tibble(
      medal = medal,
      forecast_mean = mean(forecast_draw, na.rm = TRUE),
      forecast_median = median(forecast_draw, na.rm = TRUE),
      forecast_lower_95 = quantile(forecast_draw, 0.025, na.rm = TRUE),
      forecast_upper_95 = quantile(forecast_draw, 0.975, na.rm = TRUE)
    )
  })
}

1 Overview

This report forecasts Team USA’s medal performance at the 2028 Summer Olympics in Los Angeles using historical Olympic medal data from Olympedia (“Olympedia” n.d.). The analysis follows the project prompt’s three-factor structure: Team USA’s baseline medal performance, the host-nation effect, and the effect of newly introduced Olympic sports and events (Weylandt n.d.).

The forecasting model is intentionally transparent. It starts with recent Team USA Summer Olympic results, adjusts that baseline using historical host-country performance changes, and then adds the expected medal contribution from new sports. The result is a Monte Carlo forecast for gold, silver, bronze, and total medals.

2 Data Collection

The data were collected from Olympedia by scraping Olympic editions, sport pages within each edition, and medal tables for each sport (“Olympedia” n.d.). The code uses a local cache so each Olympedia page is downloaded once and reused on later runs. This keeps the workflow reproducible and avoids repeatedly hitting the website. The first render may take a long time because uncached Olympedia pages are intentionally downloaded slowly; later renders use the saved HTML and RDS caches.

The final scraped table has one row for each Olympic edition, sport or discipline, country, and medal table entry. It includes the Olympic year, season, host city, host country code, discipline name, competitor country, and counts of gold, silver, bronze, and total medals.

3 Country Code Cleaning and Mapping

Olympedia country codes do not always match modern ISO3 country codes. Greece appears as GRE in Olympedia but GRC in ISO3, and several historical Olympic delegations no longer map cleanly to a current country. The cleaning step standardizes common mismatches so the medal data can be joined to geospatial country shapes from Natural Earth (“Natural Earth” n.d.).

Historical codes were handled with simple, explicit assumptions. Germany-related historical codes were mapped to Germany, Soviet Union-related codes were mapped to Russia, and mixed-team entries were treated as non-geographic. This is a practical data-cleaning choice for visualization and broad country-level modeling, not a political claim about historical continuity.

7 Final Fundraising Brief

Team USA enters the 2028 Los Angeles Olympics with a historically strong Summer Olympic baseline and a meaningful home-country advantage. Since 1960, Team USA has averaged approximately 0 gold medals per Summer Olympics, before accounting for the special conditions attached to hosting.

The host-nation factor is important because Olympic hosts often perform better than they did in the previous same-season Olympics. In practical terms, the Los Angeles Games should give Team USA more chances to convert near-medal performances into medals through larger participation, crowd support, and national investment around the event.

The 2028 program also creates additional upside through new events. The model uses 22 additional medal events and applies the historical host-country win rate in newly introduced sports. This adjustment is smaller than the baseline and host effects, but it still increases the expected medal count and is especially relevant because several 2028 sports are areas where the United States is likely to be competitive.

Overall, the model forecasts that Team USA should perform above its ordinary Summer Olympic baseline in Los Angeles. The exact medal count is uncertain, but the simulation gives a transparent range of plausible outcomes that can be used in fundraising and sponsor conversations. The main message for sponsors is clear: the combination of Team USA’s historical strength, home-country advantage, and new medal opportunities makes LA 2028 a strong opportunity to associate with a likely high-performing national team.
